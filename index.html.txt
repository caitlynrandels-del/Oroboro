<script>


const canvas =
document.getElementById("canvas");

const ctx =
canvas.getContext("2d");



function resize(){

canvas.width=innerWidth;

canvas.height=innerHeight;

}

resize();

window.onresize=resize;





const STATES = {


DORMANT:{
color:"#333",
energy:.1
},


AWAKENING:{
color:"#777",
energy:.5
},


ACTIVE:{
color:"#00ffff",
energy:1
},


FLOWING:{
color:"#00ff88",
energy:1.5
},


CREATIVE:{
color:"#ff00ff",
energy:2
},


CONFLICT:{
color:"#ff3300",
energy:1.8
},


STABLE:{
color:"#ffffff",
energy:.8
},


MEMORY:{
color:"#ffff00",
energy:.6
},


PORTAL:{
color:"#9900ff",
energy:2.5
},


COLLAPSED:{
color:"#111",
energy:0
}


};







class Node{


constructor(label,type="concept"){


this.id =
crypto.randomUUID();


this.label =
label;


this.type =
type;


this.state =
"DORMANT";


this.energy =
STATES[this.state].energy;


this.x =
Math.random()*canvas.width;


this.y =
Math.random()*canvas.height;


this.vx=0;
this.vy=0;


this.radius =
type==="core"?35:18;


this.age=0;


this.history=[];


this.activationHistory=[];


}



changeState(newState){


if(!STATES[newState])
return;


this.state=newState;


this.energy=
STATES[newState].energy;


this.activationHistory.push({

state:newState,

time:new Date().toLocaleTimeString()

});


}



activate(){


this.energy+=.5;


if(this.energy>.5)
this.changeState("AWAKENING");


if(this.energy>1)
this.changeState("ACTIVE");


if(this.energy>1.7)
this.changeState("CREATIVE");


}




update(){


this.age++;


this.x+=this.vx;

this.y+=this.vy;


this.vx*=.94;

this.vy*=.94;



this.energy*=.998;



if(this.energy<.2)
this.changeState("DORMANT");



if(this.age>1500 &&
this.energy>.7)

this.changeState("STABLE");



this.history.push({

x:this.x,

y:this.y

});


if(this.history.length>25)
this.history.shift();



}




draw(){



ctx.beginPath();


for(let p of this.history){

ctx.lineTo(
p.x,
p.y
);

}


ctx.strokeStyle=
"rgba(255,255,255,.05)";

ctx.stroke();





ctx.beginPath();


ctx.arc(

this.x,

this.y,

this.radius,

0,

Math.PI*2

);



ctx.fillStyle =
STATES[this.state].color;


ctx.globalAlpha=.6;


ctx.fill();


ctx.globalAlpha=1;



ctx.strokeStyle="white";

ctx.stroke();



ctx.fillStyle="#aaa";

ctx.textAlign="center";


ctx.fillText(

this.label,

this.x,

this.y+35

);



}



}







class Synapse{


constructor(a,b){

this.a=a;

this.b=b;

this.strength=Math.random();

this.packet=0;

}



pulse(){


this.packet=1;


this.b.activate();


}




update(){


let dx=this.b.x-this.a.x;

let dy=this.b.y-this.a.y;


let d=Math.sqrt(dx*dx+dy*dy);



let force=(d-180)*.00002;


this.a.vx+=dx*force;

this.a.vy+=dy*force;


this.b.vx-=dx*force;

this.b.vy-=dy*force;


}




draw(){


ctx.beginPath();


ctx.moveTo(
this.a.x,
this.a.y
);


ctx.lineTo(
this.b.x,
this.b.y
);


ctx.strokeStyle=
"rgba(255,255,255,.2)";


ctx.stroke();



if(this.packet>0){


let x=
this.a.x+
(this.b.x-this.a.x)
*(1-this.packet);



let y=
this.a.y+
(this.b.y-this.a.y)
*(1-this.packet);



ctx.beginPath();


ctx.arc(
x,
y,
5,
0,
Math.PI*2
);


ctx.fillStyle="#00ffff";

ctx.fill();


this.packet-=.03;


}



}



}







let nodes=[];

let synapses=[];



let core =
new Node(
"ORO-CORE",
"core"
);


core.x=700;

core.y=400;


core.changeState("ACTIVE");


nodes.push(core);







function addNode(name){


let n=
new Node(name);


nodes.push(n);


let s=
new Synapse(core,n);


synapses.push(s);


s.pulse();


}







function physics(){



for(let a of nodes){

for(let b of nodes){


if(a===b)
continue;


let dx=a.x-b.x;

let dy=a.y-b.y;


let d=Math.sqrt(dx*dx+dy*dy);


if(d<130){

a.vx+=dx*.0004;

a.vy+=dy*.0004;

}


}


}



for(let s of synapses)
s.update();



}








function loop(){


ctx.fillStyle="#020202";

ctx.fillRect(
0,
0,
canvas.width,
canvas.height
);



physics();



for(let s of synapses)
s.draw();



for(let n of nodes){

n.update();

n.draw();

}



requestAnimationFrame(loop);

}


loop();







function inject(){


let input =
document.getElementById("input");


let text =
input.value.trim();


if(!text)return;



if(text.includes(":")){


let parts=text.split(":");


let target=parts[0];

let state=parts[1];


let node=
nodes.find(
n=>n.label===target.toUpperCase()
);


if(node){

node.changeState(
state.toUpperCase()
);

log(
`${target} → ${state}`
);

}


}

else{


addNode(text);


log(
"Created node: "+text
);


}



input.value="";


}





function log(t){


document.getElementById("log")
.innerHTML +=
`<div class="msg">${t}</div>`;

}







canvas.onclick=e=>{


for(let n of nodes){


let dx=n.x-e.clientX;

let dy=n.y-e.clientY;


if(Math.sqrt(dx*dx+dy*dy)<n.radius){


let box=
document.getElementById("inspect");


box.style.display="block";


box.innerHTML=

`

<b>${n.label}</b><br><br>

TYPE:
${n.type}<br>

STATE:
${n.state}<br>

ENERGY:
${n.energy.toFixed(2)}

<br><br>

History:

<br>

${n.activationHistory
.map(x=>x.state)
.join("<br>")}

`;



}


}


};








function saveWorld(){


localStorage.oroboro=
JSON.stringify(nodes);


log("Universe saved");


}





function loadWorld(){


let data=
localStorage.oroboro;


if(!data)return;



nodes=[];

synapses=[];


JSON.parse(data)
.forEach(o=>{


let n =
new Node(
o.label,
o.type
);


Object.assign(n,o);


nodes.push(n);


});


log("Universe restored");


}
